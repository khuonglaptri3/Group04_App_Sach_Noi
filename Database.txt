-- =========================================================
-- BOOKS HUB DATABASE SCHEMA - COMPLETE PRODUCTION VERSION
-- Audiobooks + Ebooks + Highlights + Notes + Sync
-- PostgreSQL / Supabase
-- Bao gồm: Tạo bảng + RLS đầy đủ + Triggers + Functions
-- =========================================================

-- Enable UUID Extension
create extension if not exists "uuid-ossp";


-- =========================================================
-- 1. USER PROFILES
-- =========================================================

create table public.profiles (
    id uuid primary key references auth.users(id) on delete cascade,

    email text,
    full_name text,
    avatar_url text,
    bio text,

    is_premium boolean default false,

    books_read_count integer default 0,
    listening_hours decimal default 0,

    created_at timestamp with time zone
        default timezone('utc', now()),

    updated_at timestamp with time zone
        default timezone('utc', now())
);

-- =========================================================
-- 2. AUTHORS
-- =========================================================

create table public.authors (
    id uuid primary key default uuid_generate_v4(),

    name text not null,
    biography text,
    avatar_url text,

    created_at timestamp with time zone
        default timezone('utc', now())
);

-- =========================================================
-- 3. CATEGORIES
-- =========================================================

create table public.categories (
    id uuid primary key default uuid_generate_v4(),

    name_vi text not null,
    name_en text not null,

    icon_url text,

    gradient_colors text[],

    created_at timestamp with time zone
        default timezone('utc', now())
);

-- =========================================================
-- 4. BOOKS
-- =========================================================

create table public.books (
    id uuid primary key default uuid_generate_v4(),

    title text not null,
    description text,

    cover_url text,
    banner_url text,

    language text default 'vi',

    rating_avg decimal default 0,
    rating_count integer default 0,

    price decimal default 0,

    is_audiobook boolean default true,
    is_ebook boolean default true,

    is_premium_only boolean default false,

    duration_seconds integer,
    page_count integer,

    published_year integer,

    author_id uuid
        references public.authors(id),

    category_id uuid
        references public.categories(id),

    created_at timestamp with time zone
        default timezone('utc', now())
);

-- =========================================================
-- 5. BOOK FILES
-- =========================================================

create table public.book_files (
    id uuid primary key default uuid_generate_v4(),

    book_id uuid
        references public.books(id)
        on delete cascade,

    file_type text not null,
    -- epub / pdf / mp3 / m4b

    file_url text not null,

    file_size_mb decimal,

    created_at timestamp with time zone
        default timezone('utc', now())
);

-- =========================================================
-- 6. BOOK CHAPTERS
-- =========================================================

create table public.book_chapters (
    id uuid primary key default uuid_generate_v4(),

    book_id uuid
        references public.books(id)
        on delete cascade,

    chapter_index integer not null,

    title text,

    start_time_seconds integer,
    end_time_seconds integer,

    start_page integer,
    end_page integer,

    created_at timestamp with time zone
        default timezone('utc', now())
);

-- =========================================================
-- 7. USER LIBRARY
-- =========================================================

create table public.user_library (
    id uuid primary key default uuid_generate_v4(),

    user_id uuid
        references public.profiles(id)
        on delete cascade,

    book_id uuid
        references public.books(id)
        on delete cascade,

    is_purchased boolean default false,
    is_downloaded boolean default false,
    is_favorite boolean default false,

    purchased_at timestamp with time zone,

    last_accessed timestamp with time zone
        default timezone('utc', now()),

    unique(user_id, book_id)
);

-- =========================================================
-- 8. READING / LISTENING PROGRESS
-- =========================================================

create table public.progress (
    id uuid primary key default uuid_generate_v4(),

    user_id uuid
        references public.profiles(id)
        on delete cascade,

    book_id uuid
        references public.books(id)
        on delete cascade,

    percent_complete integer
        default 0
        check (percent_complete between 0 and 100),

    current_chapter_id uuid
        references public.book_chapters(id),

    -- audiobook
    last_position_seconds integer default 0,

    -- ebook
    last_page_number integer default 0,

    -- epub support
    epub_cfi text,

    updated_at timestamp with time zone
        default timezone('utc', now()),

    unique(user_id, book_id)
);

-- =========================================================
-- 9. BOOKMARKS

create table public.bookmarks (

-- =========================================================

-- A. Tự động tạo hồ sơ & streak khi người dùng mới đăng ký
create or replace function public.handle_new_user()
returns trigger as $$
begin
    insert into public.profiles (id, email, full_name, avatar_url, is_premium)
    values (
        new.id,
        new.email,
        coalesce(
            new.raw_user_meta_data->>'full_name',
            new.raw_user_meta_data->>'name',
            'User_' || substr(new.id::text, 1, 8)
        ),
        new.raw_user_meta_data->>'avatar_url',
        false
    );

    insert into public.reading_streaks (user_id, current_streak, longest_streak, last_read_date)
    values (new.id, 0, 0, null);

    return new;
end;
$$ language plpgsql security definer;

-- B. Tự động tính lại trung bình đánh giá của sách
create or replace function public.update_book_ratings()
returns trigger as $$
declare
    v_book_id uuid;
    v_avg_rating decimal;
    v_rating_count integer;
begin
    if (tg_op = 'DELETE') then
        v_book_id := old.book_id;
    else
        v_book_id := new.book_id;
    end if;

    select count(*), coalesce(avg(rating), 0)
    into v_rating_count, v_avg_rating
    from public.reviews
    where book_id = v_book_id;

    update public.books
    set rating_avg   = round(v_avg_rating, 2),
        rating_count = v_rating_count
    where id = v_book_id;

    return null;
end;
$$ language plpgsql security definer;

-- C. Tự động cập nhật chỉ mục tìm kiếm (search_vector)
create or replace function public.sync_search_vector()
returns trigger as $$
begin
    new.search_vector :=
        to_tsvector('simple', coalesce(new.title, '')) ||
        to_tsvector('simple', coalesce(new.description, ''));
    return new;
end;
$$ language plpgsql;

-- D. Cập nhật tiến độ đọc, chuỗi streak, giờ nghe, và thư viện
create or replace function public.handle_progress_update()
returns trigger as $$
declare
    v_today        date;
    v_last_read    date;
    v_curr_streak  int;
    v_is_audiobook boolean;
    v_pos_diff     integer;
begin
    v_today := (timezone('utc', now()))::date;

    -- 1. Cập nhật Streak đọc sách
    select last_read_date, current_streak
    into v_last_read, v_curr_streak
    from public.reading_streaks
    where user_id = new.user_id;

    if not found then
        insert into public.reading_streaks (user_id, current_streak, longest_streak, last_read_date)
        values (new.user_id, 1, 1, v_today);
    else
        if v_last_read is null then
            update public.reading_streaks
            set current_streak = 1,
                longest_streak = greatest(longest_streak, 1),
                last_read_date = v_today
            where user_id = new.user_id;
        elsif v_last_read = v_today - 1 then
            update public.reading_streaks
            set current_streak = current_streak + 1,
                longest_streak = greatest(longest_streak, current_streak + 1),
                last_read_date = v_today
            where user_id = new.user_id;
        elsif v_last_read < v_today - 1 then
            update public.reading_streaks
            set current_streak = 1,
                longest_streak = greatest(longest_streak, 1),
                last_read_date = v_today
            where user_id = new.user_id;
        end if;
        -- Nếu v_last_read = v_today thì bỏ qua (đã đọc hôm nay rồi)
    end if;

    -- 2. Tăng books_read_count khi hoàn thành (100%)
    if new.percent_complete = 100
       and (tg_op = 'INSERT' or (tg_op = 'UPDATE' and old.percent_complete < 100))
    then
        update public.profiles
        set books_read_count = books_read_count + 1
        where id = new.user_id;
    end if;

    -- 3. Tích lũy số giờ nghe đối với audiobook
    select is_audiobook into v_is_audiobook
    from public.books
    where id = new.book_id;

    if v_is_audiobook = true then
        v_pos_diff := 0;
        if tg_op = 'UPDATE' then
            if new.last_position_seconds > old.last_position_seconds then
                v_pos_diff := new.last_position_seconds - old.last_position_seconds;
            end if;
        elsif tg_op = 'INSERT' then
            v_pos_diff := new.last_position_seconds;
        end if;

        if v_pos_diff > 0 then
            update public.profiles
            set listening_hours = listening_hours + (v_pos_diff::decimal / 3600.0)
            where id = new.user_id;
        end if;
    end if;

    -- 4. Cập nhật last_accessed trong user_library
    insert into public.user_library (user_id, book_id, last_accessed)
    values (new.user_id, new.book_id, timezone('utc', now()))
    on conflict (user_id, book_id) do update
    set last_accessed = timezone('utc', now());

    return new;
end;
$$ language plpgsql security definer;

-- E. Tự động mở khóa Huy hiệu dựa trên thông số Đọc/Nghe
create or replace function public.check_and_award_badges()
returns trigger as $$
declare
    v_has_badge boolean;
begin
    -- Huy hiệu 1: Độc giả Khởi đầu (>= 1 cuốn)
    if new.books_read_count >= 1 then
        select exists(
            select 1 from public.user_badges
            where user_id = new.id
              and badge_id = 'b0000000-0000-0000-0000-000000000001'
        ) into v_has_badge;
        if not v_has_badge then
            insert into public.user_badges (user_id, badge_id, unlocked_at)
            values (new.id, 'b0000000-0000-0000-0000-000000000001', timezone('utc', now()))
            on conflict do nothing;
        end if;
    end if;

    -- Huy hiệu 2: Mọt sách Chính hiệu (>= 5 cuốn)
    if new.books_read_count >= 5 then
        select exists(
            select 1 from public.user_badges
            where user_id = new.id
              and badge_id = 'b0000000-0000-0000-0000-000000000002'
        ) into v_has_badge;
        if not v_has_badge then
            insert into public.user_badges (user_id, badge_id, unlocked_at)
            values (new.id, 'b0000000-0000-0000-0000-000000000002', timezone('utc', now()))
            on conflict do nothing;
        end if;
    end if;

    -- Huy hiệu 3: Đại danh hào (>= 20 cuốn)
    if new.books_read_count >= 20 then
        select exists(
            select 1 from public.user_badges
            where user_id = new.id
              and badge_id = 'b0000000-0000-0000-0000-000000000003'
        ) into v_has_badge;
        if not v_has_badge then
            insert into public.user_badges (user_id, badge_id, unlocked_at)
            values (new.id, 'b0000000-0000-0000-0000-000000000003', timezone('utc', now()))
            on conflict do nothing;
        end if;
    end if;

    -- Huy hiệu 4: Thính giả Chăm chỉ (>= 10 giờ)
    if new.listening_hours >= 10 then
        select exists(
            select 1 from public.user_badges
            where user_id = new.id
              and badge_id = 'b0000000-0000-0000-0000-000000000004'
        ) into v_has_badge;
        if not v_has_badge then
            insert into public.user_badges (user_id, badge_id, unlocked_at)
            values (new.id, 'b0000000-0000-0000-0000-000000000004', timezone('utc', now()))
            on conflict do nothing;
        end if;
    end if;

    -- Huy hiệu 5: Siêu cấp Thính giả (>= 50 giờ)
    if new.listening_hours >= 50 then
        select exists(
            select 1 from public.user_badges
            where user_id = new.id
              and badge_id = 'b0000000-0000-0000-0000-000000000005'
        ) into v_has_badge;
        if not v_has_badge then
            insert into public.user_badges (user_id, badge_id, unlocked_at)
            values (new.id, 'b0000000-0000-0000-0000-000000000005', timezone('utc', now()))
            on conflict do nothing;
        end if;
    end if;

    return new;
end;
$$ language plpgsql security definer;

-- 1. Tìm kiếm sách nâng cao (Full-Text Search + bộ lọc)
create or replace function public.search_books(
    query_text      text,
    category_filter uuid    default null,
    author_filter   uuid    default null,
    limit_val       integer default 20,
    offset_val      integer default 0
)
returns table (
    id               uuid,
    title            text,
    description      text,
    cover_url        text,
    banner_url       text,
    language         text,
    rating_avg       decimal,
    rating_count     integer,
    price            decimal,
    is_audiobook     boolean,
    is_ebook         boolean,
    is_premium_only  boolean,
    duration_seconds integer,
    page_count       integer,
    published_year   integer,
    author_id        uuid,
    author_name      text,
    category_id      uuid,
    category_name_vi text,
    category_name_en text,
    similarity_rank  real
) as $$
    select
        b.id,
        b.title,
        b.description,
        b.cover_url,
        b.banner_url,
        b.language,
        b.rating_avg,
        b.rating_count,
        b.price,
        b.is_audiobook,
        b.is_ebook,
        b.is_premium_only,
        b.duration_seconds,
        b.page_count,
        b.published_year,
        b.author_id,
        a.name            as author_name,
        b.category_id,
        c.name_vi         as category_name_vi,
        c.name_en         as category_name_en,
        case
            when query_text is not null and query_text <> ''
            then ts_rank(b.search_vector, plainto_tsquery('simple', query_text))
            else 1.0::real
        end               as similarity_rank
    from public.books b
    left join public.authors    a on b.author_id   = a.id
    left join public.categories c on b.category_id = c.id
    where
        (category_filter is null or b.category_id = category_filter)
        and (author_filter is null or b.author_id = author_filter)
        and (
            query_text is null or query_text = '' or
            b.search_vector @@ plainto_tsquery('simple', query_text) or
            b.title ilike '%' || query_text || '%'
        )
    order by
        case when query_text is not null and query_text <> '' then 1 else 0 end desc,
        similarity_rank desc,
        b.rating_avg desc,
        b.rating_count desc
    limit limit_val
    offset offset_val;
$$ language sql stable;

-- 2. Gợi ý sách tương tự (cùng thể loại hoặc tác giả)
create or replace function public.get_similar_books(
    target_book_id uuid,
    limit_val      integer default 5
)
returns table (
    id              uuid,
    title           text,
    cover_url       text,
    rating_avg      decimal,
    rating_count    integer,
    is_audiobook    boolean,
    is_ebook        boolean,
    is_premium_only boolean,
    price           decimal,
    author_name     text
) as $$
    select
        b.id,
        b.title,
        b.cover_url,
        b.rating_avg,
        b.rating_count,
        b.is_audiobook,
        b.is_ebook,
        b.is_premium_only,
        b.price,
        a.name as author_name
    from public.books b
    left join public.authors a on b.author_id = a.id
    cross join (
        select category_id, author_id
        from public.books
        where id = target_book_id
    ) target
    where b.id <> target_book_id
      and (b.category_id = target.category_id or b.author_id = target.author_id)
    order by
        case when b.author_id   = target.author_id   then 2 else 0 end +
        case when b.category_id = target.category_id then 1 else 0 end desc,
        b.rating_avg desc,
        b.rating_count desc
    limit limit_val;
$$ language sql stable;

-- 3. Tổng hợp thống kê đọc + huy hiệu (dạng JSON) cho màn hình Profile
create or replace function public.get_user_reading_summary(
    target_user_id uuid
)
returns json as $$
declare
    v_books_read      integer;
    v_listening_hours decimal;
    v_current_streak  integer;
    v_longest_streak  integer;
    v_total_badges    integer;
    v_reading_count   integer;
    v_purchased_count integer;
    v_result          json;
begin
    select books_read_count, listening_hours
    into v_books_read, v_listening_hours
    from public.profiles
    where id = target_user_id;

    select current_streak, longest_streak
    into v_current_streak, v_longest_streak
    from public.reading_streaks
    where user_id = target_user_id;

    select count(*) into v_total_badges
    from public.user_badges
    where user_id = target_user_id;

    select count(*) into v_reading_count
    from public.progress
    where user_id = target_user_id
      and percent_complete > 0
      and percent_complete < 100;

    select count(*) into v_purchased_count
    from public.user_library
    where user_id = target_user_id
      and is_purchased = true;

    v_result := json_build_object(
        'books_read_count',       coalesce(v_books_read, 0),
        'listening_hours',        coalesce(v_listening_hours, 0.0),
        'current_streak',         coalesce(v_current_streak, 0),
        'longest_streak',         coalesce(v_longest_streak, 0),
        'total_badges_unlocked',  coalesce(v_total_badges, 0),
        'books_currently_reading',coalesce(v_reading_count, 0),
        'books_purchased',        coalesce(v_purchased_count, 0)
    );

    return v_result;
end;
$$ language plpgsql stable security definer;

-- 4. Bật/Tắt yêu thích một cuốn sách
create or replace function public.toggle_favorite_book(
    user_uuid uuid,
    book_uuid uuid
)
returns boolean as $$
declare
    v_current_fav boolean;
begin
    select is_favorite into v_current_fav
    from public.user_library
    where user_id = user_uuid and book_id = book_uuid;

    if not found then
        insert into public.user_library (user_id, book_id, is_favorite, last_accessed)
        values (user_uuid, book_uuid, true, timezone('utc', now()));
        return true;
    else
        update public.user_library
        set is_favorite  = not v_current_fav,
            last_accessed = timezone('utc', now())
        where user_id = user_uuid and book_id = book_uuid;
        return not v_current_fav;
    end if;
end;
$$ language plpgsql security definer;

create or replace function public.complete_payment(
    payment_uuid uuid
)
returns void as $$
declare
    v_user_id uuid;
    v_book_id uuid;
    v_payment_type text;
begin

    select
        user_id,
        book_id,
        payment_type
    into
        v_user_id,
        v_book_id,
        v_payment_type
    from public.payments
    where id = payment_uuid;

    if not found then
        raise exception 'Payment not found';
    end if;


    update public.payments
    set
        status = 'paid',
        paid_at = timezone('utc', now()),
        updated_at = timezone('utc', now())
    where id = payment_uuid;


    update public.invoices
    set
        status = 'paid',
        paid_at = timezone('utc', now())
    where payment_id = payment_uuid;


    if v_payment_type = 'book_purchase' then

        perform public.purchase_book(
            v_user_id,
            v_book_id
        );

        insert into public.notifications (
            user_id,
            type,
            data
        )
        values (
            v_user_id,
            'book_unlocked',
            jsonb_build_object(
                'book_id', v_book_id,
                'payment_id', payment_uuid
            )
        );

    elsif v_payment_type = 'premium_subscription' then

        update public.profiles
        set
            is_premium = true,
            updated_at = timezone('utc', now())
        where id = v_user_id;

        insert into public.notifications (
            user_id,
            type,
            data
        )
        values (
            v_user_id,
            'premium_activated',
            jsonb_build_object(
                'payment_id', payment_uuid
            )
        );

    end if;

end;
$$ language plpgsql security definer;

-- 1. Function tự tính lại rating_avg và rating_count
create or replace function public.update_book_ratings()
returns trigger as $$
declare
    v_book_id uuid;
    v_avg_rating numeric;
    v_rating_count integer;
begin
    if tg_op = 'DELETE' then
        v_book_id := old.book_id;
    else
        v_book_id := new.book_id;
    end if;

    select
        count(*),
        coalesce(avg(rating)::numeric, 0)
    into
        v_rating_count,
        v_avg_rating
    from public.reviews
    where book_id = v_book_id;

    update public.books
    set
        rating_avg = round(v_avg_rating, 2),
        rating_count = v_rating_count
    where id = v_book_id;

    return null;
end;
$$ language plpgsql security definer;
